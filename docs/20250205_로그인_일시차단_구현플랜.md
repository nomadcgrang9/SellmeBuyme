## 📁 수정 대상 파일

### 1. `src/lib/constants/adminEmails.ts` (신규)

```typescript
/**
 * 관리자 이메일 화이트리스트
 * 개인정보처리방침 준비 전까지 로그인 허용 대상
 */
export const ADMIN_WHITELIST_EMAILS = [
  'l30417305@gmail.com',
];

/**
 * 화이트리스트 포함 여부 확인
 */
export function isWhitelistedAdmin(email: string | null | undefined): boolean {
  if (!email) return false;
  return ADMIN_WHITELIST_EMAILS.includes(email.toLowerCase());
}
```

### 2. `src/pages/AuthCallback.tsx` (수정)

**추가 Import**:
```typescript
import { isWhitelistedAdmin } from '@/lib/constants/adminEmails';
import { useToastStore } from '@/stores/toastStore';
```

**수정 위치**: `exchangeCodeForSession` 호출 후

```typescript
if (session) {
  const { user } = session;

  // ⭐ 화이트리스트 체크
  if (!isWhitelistedAdmin(user.email)) {
    await supabase.auth.signOut();

    useToastStore.getState().addToast({
      type: 'error',
      message: '개인정보처리방침 준비 중',
      description: '현재 로그인 기능은 임시로 제한되어 있습니다. 빠른 시일 내에 정상 서비스를 제공하겠습니다.',
      duration: 5000,
    });

    navigate('/', { replace: true });
    return;
  }

  // 관리자는 정상 진행
  await createMinimalProfile(user.id, user.email || 'unknown');
  navigate('/', { replace: true });
}
```
